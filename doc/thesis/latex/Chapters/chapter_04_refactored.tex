\section{Classical Petri Net Review}
\label{sec:classical-petri-net-review}

\subsection{Basic Definitions}
\label{subsec:basic-definitions}

A \emph{classical Petri net} is a 5-tuple:
%
\begin{equation}
\text{PN} = (P, T, F, W, M_0)
\end{equation}
%
where:
\begin{description}
    \item[$P = \{p_1, p_2, \ldots, p_m\}$] is a finite set of \emph{places} (represented as circles)
    \item[$T = \{t_1, t_2, \ldots, t_n\}$] is a finite set of \emph{transitions} (represented as rectangles or bars)
    \item[$F \subseteq (P \times T) \cup (T \times P)$] is the \emph{flow relation} (directed arcs)
    \item[$W: F \to \mathbb{N}^+$] is the \emph{arc weight function} (positive integers, default $W = 1$)
    \item[$M_0: P \to \mathbb{N}_0$] is the \emph{initial marking} (token distribution)
\end{description}
%
with the constraint: $P \cap T = \emptyset$ (places and transitions are disjoint).

\subsection{Neighborhood Notation}
\label{subsec:neighborhood-notation}

For a transition $t \in T$:
\begin{align}
{}^\bullet t &= \{p \in P \mid (p,t) \in F\} && \text{(\emph{preset}: input places)} \\
t^\bullet &= \{p \in P \mid (t,p) \in F\} && \text{(\emph{postset}: output places)}
\end{align}

For a place $p \in P$:
\begin{align}
{}^\bullet p &= \{t \in T \mid (t,p) \in F\} && \text{(transitions producing tokens in $p$)} \\
p^\bullet &= \{t \in T \mid (p,t) \in F\} && \text{(transitions consuming tokens from $p$)}
\end{align}

\subsection{Enabling Rule}
\label{subsec:enabling-rule}

A transition $t$ is \emph{enabled} at marking $M$, written $M[t\rangle$, if and only if:
%
\begin{equation}
\forall p \in {}^\bullet t: M(p) \geq W(p,t)
\end{equation}
%
That is, all input places must contain at least $W(p,t)$ tokens.

\subsection{Firing Rule}
\label{subsec:firing-rule}

When an enabled transition $t$ fires, the new marking $M'$ is computed as:
%
\begin{equation}
\forall p \in P: M'(p) = M(p) - W(p,t) + W(t,p)
\end{equation}
%
where:
\begin{itemize}
    \item $W(p,t) = 0$ if $(p,t) \notin F$ (no arc from $p$ to $t$)
    \item $W(t,p) = 0$ if $(t,p) \notin F$ (no arc from $t$ to $p$)
\end{itemize}

\noindent\textbf{Notation:} $M[t\rangle M'$ denotes ``transition $t$ is enabled at $M$ and produces $M'$''.

\subsection{Reachability}
\label{subsec:reachability}

The \emph{reachability set} $R(M_0)$ is the set of all markings reachable from $M_0$:
%
\begin{equation}
R(M_0) = \{M \mid \exists \sigma = t_1 t_2 \cdots t_k: M_0[\sigma\rangle M\}
\end{equation}
%
where $\sigma$ is a \emph{firing sequence} of transitions.

\subsection{Limitations for Biological Modeling}
\label{subsec:limitations-biological-modeling}

Classical Petri nets face fundamental limitations when modeling biological systems:

\begin{enumerate}
\item \textbf{All arcs are consumptive}: Tokens are always removed from input places
    \begin{itemize}
        \item \textbf{Problem}: Enzymes catalyze reactions without being consumed
        \item \textbf{Example}: Hexokinase phosphorylates glucose but remains unchanged
        \item If modeled with normal arcs: enzyme would deplete after first reaction
    \end{itemize}

\item \textbf{No regulatory inhibition}: Transitions fire whenever inputs are available
    \begin{itemize}
        \item \textbf{Problem}: Biological reactions can be blocked by inhibitors
        \item \textbf{Example}: High ATP inhibits phosphofructokinase (energy feedback)
        \item Classical PNs cannot express threshold-based blocking
    \end{itemize}

\item \textbf{Strong independence requirement}: Parallel execution requires disjoint neighborhoods
    \begin{itemize}
        \item \textbf{Problem}: Biological reactions share places (metabolites, enzymes)
        \item \textbf{Example}: Multiple reactions produce same metabolite (superposition)
        \item Classical parallelism rejects most biological networks
    \end{itemize}

\item \textbf{Homogeneous transitions}: All transitions follow same firing semantics
    \begin{itemize}
        \item \textbf{Problem}: Biology exhibits multiple temporal scales
        \item \textbf{Example}: Enzyme kinetics (continuous) vs gene expression (stochastic bursts)
        \item Classical PNs cannot mix continuous and discrete dynamics natively
    \end{itemize}

\item \textbf{Abstract tokens}: No semantic meaning beyond quantity
    \begin{itemize}
        \item \textbf{Problem}: Biochemistry requires elemental composition tracking
        \item \textbf{Example}: \ce{C6H12O6 + ATP -> C6H11O9P + ADP} (atoms must balance)
        \item Classical PNs cannot validate stoichiometric correctness
    \end{itemize}
\end{enumerate}

These limitations motivate the \emph{Extended Biological Petri Net} formalism presented in this chapter.

\section{Extended Bio-PN: The 12-Tuple Definition}
\label{sec:extended-biopn-12tuple}
%************************************************

An \emph{Extended Biological Petri Net} is a 12-tuple:
%
\begin{equation}
\text{BioPN} = (P, T, F, W, M_0, K, \Phi, \Sigma, \Theta, \Delta, \tau, \rho)
\end{equation}

\subsection{Core Components (From Classical PN)}
\label{subsec:core-components}

\paragraph{P: Places (Biochemical Species)}
Places represent biochemical entities:
\begin{itemize}
    \item Metabolites: Glucose, ATP, \ce{NAD+}, Pyruvate
    \item Enzymes: Hexokinase, Phosphofructokinase, Pyruvate Kinase
    \item Genes: \textit{lacZ}, \textit{lacY}, \textit{lacA} (structural genes)
    \item Transcription Factors: CRP, cAMP-CRP complex, LacI repressor
    \item mRNA: \textit{lacZ}\_mRNA, \textit{lacY}\_mRNA
    \item Proteins: $\beta$-galactosidase, Permease, Transacetylase
\end{itemize}

Each place $p$ has attributes:
\begin{description}
    \item[\texttt{p.id}] Unique identifier (e.g., ``P001'')
    \item[\texttt{p.name}] Human-readable name (e.g., ``Glucose'', ``ATP'')
    \item[\texttt{p.formula}] Biochemical formula (e.g., ``\ce{C6H12O6}'', ``\ce{C10H16N5O13P3}'')
    \item[\texttt{p.type}] $\in$ \{\texttt{metabolite}, \texttt{enzyme}, \texttt{gene}, \texttt{mRNA}, \texttt{protein}, \texttt{complex}\}
\end{description}

\paragraph{T: Transitions (Biochemical Reactions)}
Transitions represent biochemical transformations:
\begin{itemize}
    \item Enzymatic reactions: Hexokinase, PGI, PFK, Aldolase, etc.
    \item Transcription: Gene $\to$ mRNA
    \item Translation: mRNA $\to$ Protein
    \item Complex formation: cAMP + CRP $\to$ cAMP-CRP
    \item Degradation: mRNA $\to \emptyset$, Protein $\to \emptyset$
\end{itemize}

Each transition $t$ has attributes:
\begin{description}
    \item[\texttt{t.id}] Unique identifier (e.g., ``T001'')
    \item[\texttt{t.name}] Reaction name (e.g., ``Hexokinase'', ``\textit{lacZ}\_transcription'')
    \item[\texttt{t.ec\_number}] Enzyme Commission number (e.g., ``2.7.1.1'' for hexokinase)
    \item[\texttt{t.reversible}] Boolean (\texttt{true} for reversible reactions)
\end{description}

\paragraph{F $\subseteq$ (P $\times$ T) $\cup$ (T $\times$ P): Normal Arcs}
Normal arcs represent stoichiometric flow:
\begin{itemize}
    \item $(p,t) \in F$: place $p$ is substrate (consumed by $t$)
    \item $(t,p) \in F$: place $p$ is product (produced by $t$)
\end{itemize}

\paragraph{W: F $\to \mathbb{N}^+$: Arc Weights (Stoichiometric Coefficients)}
\begin{itemize}
    \item $W(p,t)$ = number of tokens consumed from $p$ when $t$ fires
    \item $W(t,p)$ = number of tokens produced in $p$ when $t$ fires
    \item Default: $W = 1$ (one molecule consumed/produced)
    \item Example: Hexokinase consumes 1 Glucose + 1 ATP, produces 1 G6P + 1 ADP
\end{itemize}

\paragraph{$M_0$: P $\to \mathbb{N}_0 \cup \mathbb{R}_0^+$: Initial Marking}
\begin{itemize}
    \item Discrete ($\mathbb{N}_0$): Integer token counts for stochastic models
        \begin{itemize}
            \item Example: $M_0(\text{Glucose}) = 1000$ molecules
        \end{itemize}
    \item Continuous ($\mathbb{R}_0^+$): Real-valued concentrations for ODE models
        \begin{itemize}
            \item Example: $M_0(\text{Glucose}) = 5.0$ mM (millimolar)
        \end{itemize}
\end{itemize}

\subsection{Extended Components (Novel to Bio-PN)}
\label{subsec:extended-components}

\paragraph{K: Capacity Function (Optional Bounds)}
$K: P \to \mathbb{N} \cup \{\infty\}$ assigns maximum token capacity to places.
\begin{itemize}
    \item Default: $K(p) = \infty$ (unbounded)
    \item Biological use: Cellular compartment saturation, enzyme availability limits
    \item Example: $K(\text{Enzyme}) = 100$ (only 100 enzyme molecules available)
\end{itemize}

\paragraph{$\Phi$: Rate Function (Kinetic Laws)}
$\Phi: T \to$ \{\texttt{KineticLaw}\} assigns rate expressions to transitions.

\noindent\textbf{Format}:
\begin{itemize}
    \item Continuous transitions: ODE-based rates
        \begin{itemize}
            \item Michaelis-Menten: $\Phi(t) = \frac{V_{\max} [S]}{K_m + [S]}$
            \item Mass action: $\Phi(t) = k \prod_{p \in {}^\bullet t} M(p)$
        \end{itemize}
    \item Stochastic transitions: Propensity functions
        \begin{itemize}
            \item Gene expression burst: $\Phi(t) = k_{\text{burst}} \cdot M(\text{Gene})$
        \end{itemize}
\end{itemize}

\paragraph{$\Sigma$: Regulatory Structure (Test \& Inhibitor Arcs)}
$\Sigma \subseteq (P \times T) \setminus F$ defines non-consumptive regulatory relationships.

\noindent\textbf{Test Arcs} (Catalysis): $(p,t) \in \Sigma_{\text{test}}$
\begin{itemize}
    \item \textbf{Semantics}: Place $p$ must have tokens for $t$ to fire, but tokens are \emph{not consumed}
    \item \textbf{Biological meaning}: $p$ is an enzyme/catalyst for reaction $t$
    \item \textbf{Example}: Hexokinase enables glucose phosphorylation without being consumed
    \item \textbf{Enabling condition}: $M(p) \geq \theta(p,t)$ where $\theta$ is a threshold (typically 1)
\end{itemize}

\noindent\textbf{Inhibitor Arcs} (Feedback Inhibition): $(p,t) \in \Sigma_{\text{inhibit}}$
\begin{itemize}
    \item \textbf{Semantics}: If $M(p) \geq \theta(p,t)$, transition $t$ is \emph{disabled}
    \item \textbf{Biological meaning}: High concentration of $p$ blocks reaction $t$
    \item \textbf{Example}: High ATP inhibits PFK (glycolysis feedback control)
    \item \textbf{Disabling condition}: $M(p) \geq \theta_{\text{inhibit}}(p,t)$
\end{itemize}

\paragraph{$\Theta$: Environmental Exchange Classification}
$\Theta: T \to \{\text{INTERNAL}, \text{SOURCE}, \text{SINK}, \text{EXCHANGE}\}$

\begin{description}
    \item[INTERNAL] Standard reaction: ${}^\bullet t \neq \emptyset$ and $t^\bullet \neq \emptyset$
    \item[SOURCE] External input: ${}^\bullet t = \emptyset$ and $t^\bullet \neq \emptyset$
        \begin{itemize}
            \item Example: Glucose import from bloodstream
        \end{itemize}
    \item[SINK] External output: ${}^\bullet t \neq \emptyset$ and $t^\bullet = \emptyset$
        \begin{itemize}
            \item Example: CO$_2$ export to environment
        \end{itemize}
    \item[EXCHANGE] Bidirectional transport: Reversible membrane transport
\end{description}

\paragraph{$\Delta$: Dependency Taxonomy}
$\Delta: T \times T \to \{\text{INDEPENDENT}, \text{CONVERGENT}, \text{REGULATORY}, \text{COMPETITIVE}\}$

Classification of transition pairs for parallel execution:
\begin{description}
    \item[INDEPENDENT] Disjoint neighborhoods: $({}^\bullet t_1 \cup t_1^\bullet) \cap ({}^\bullet t_2 \cup t_2^\bullet) = \emptyset$
    \item[CONVERGENT] Share output places only: $t_1^\bullet \cap t_2^\bullet \neq \emptyset$, ${}^\bullet t_1 \cap {}^\bullet t_2 = \emptyset$
    \item[REGULATORY] Share regulatory places (via $\Sigma$): Non-consumptive coupling
    \item[COMPETITIVE] Share input places: ${}^\bullet t_1 \cap {}^\bullet t_2 \neq \emptyset$ (resource conflict)
\end{description}

\paragraph{$\tau$: Transition Type}
$\tau: T \to \{\text{CONTINUOUS}, \text{STOCHASTIC}, \text{TIMED}, \text{IMMEDIATE}\}$

\begin{description}
    \item[CONTINUOUS] ODE-based dynamics (enzyme kinetics)
    \item[STOCHASTIC] Gillespie SSA (gene expression bursts)
    \item[TIMED] Fixed delay (transport, transcription elongation)
    \item[IMMEDIATE] Instantaneous firing (complex formation/dissociation)
\end{description}

\paragraph{$\rho$: Formula Mapping (Atom Tracking)}
$\rho: P \to \text{Formula}$ assigns biochemical formulas to places.

\begin{itemize}
    \item Format: $\rho(p) = \text{C}_a \text{H}_b \text{O}_c \text{N}_d \text{P}_e \text{S}_f \cdots$
    \item Example: $\rho(\text{Glucose}) = \text{C}_6\text{H}_{12}\text{O}_6$
    \item Example: $\rho(\text{ATP}) = \text{C}_{10}\text{H}_{16}\text{N}_5\text{O}_{13}\text{P}_3$
    \item Purpose: Validate mass balance across reactions
\end{itemize}

\section{Extended Enabling and Firing Rules}
\label{sec:extended-enabling-firing}
%************************************************

\subsection{Extended Enabling Rule}
\label{subsec:extended-enabling-rule}

A transition $t$ is \emph{enabled} in Extended Bio-PN at marking $M$, written $M[t\rangle$, if \textbf{all} of the following conditions hold:

\begin{enumerate}
\item \textbf{Normal arc condition} (substrate availability):
    \begin{equation}
    \forall p \in {}^\bullet t: M(p) \geq W(p,t)
    \end{equation}

\item \textbf{Test arc condition} (catalyst availability):
    \begin{equation}
    \forall (p,t) \in \Sigma_{\text{test}}: M(p) \geq \theta_{\text{test}}(p,t)
    \end{equation}

\item \textbf{Inhibitor arc condition} (no blocking):
    \begin{equation}
    \forall (p,t) \in \Sigma_{\text{inhibit}}: M(p) < \theta_{\text{inhibit}}(p,t)
    \end{equation}

\item \textbf{Capacity constraint} (output places not full):
    \begin{equation}
    \forall p \in t^\bullet: M(p) + W(t,p) \leq K(p)
    \end{equation}
\end{enumerate}

\subsection{Extended Firing Rule}
\label{subsec:extended-firing-rule}

When transition $t$ fires at marking $M$, producing $M'$:

\begin{enumerate}
\item \textbf{Normal arcs}: Consume inputs, produce outputs
    \begin{equation}
    \forall p \in P: M'(p) = M(p) - W(p,t) + W(t,p)
    \end{equation}

\item \textbf{Test arcs}: No token change (read-only)
    \begin{equation}
    \forall (p,t) \in \Sigma_{\text{test}}: M'(p) = M(p)
    \end{equation}

\item \textbf{Inhibitor arcs}: No token change (blocking only)
    \begin{equation}
    \forall (p,t) \in \Sigma_{\text{inhibit}}: M'(p) = M(p)
    \end{equation}
\end{enumerate}

\subsection{Atomic Balance Validation}
\label{subsec:atomic-balance}

For each transition $t$, we verify mass conservation:
%
\begin{equation}
\sum_{p \in {}^\bullet t} W(p,t) \cdot \rho(p) = \sum_{p \in t^\bullet} W(t,p) \cdot \rho(p)
\end{equation}
%
where $\rho(p)$ is the element vector for place $p$.

\section{Summary}
\label{sec:chapter4-summary}
%************************************************

This chapter introduced the \textbf{Extended Biological Petri Net} formalism as a 12-tuple:
%
\[
\text{BioPN} = (P, T, F, W, M_0, K, \Phi, \Sigma, \Theta, \Delta, \tau, \rho)
\]

\noindent\textbf{Key innovations}:
\begin{enumerate}
    \item \textbf{Test arcs} ($\Sigma_{\text{test}}$): Non-consumptive catalysis
    \item \textbf{Inhibitor arcs} ($\Sigma_{\text{inhibit}}$): Feedback regulation
    \item \textbf{Heterogeneous transitions} ($\tau$): Mixed dynamics (continuous + stochastic)
    \item \textbf{Biochemical formulas} ($\rho$): Atomic balance validation
    \item \textbf{Dependency taxonomy} ($\Delta$): Weak independence for parallelism
\end{enumerate}

These extensions enable accurate modeling of biological systems while preserving Petri net formal properties. The next chapter (Chapter 5) develops the theory of \emph{weak independence} for parallel execution.
