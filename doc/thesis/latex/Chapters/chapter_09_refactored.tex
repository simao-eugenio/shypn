\chapter{KEGG Compound and Reaction Integration}
\label{ch:kegg}

\section{Introduction}
\label{sec:kegg-intro}

\textbf{Biochemical formula tracking} (Chapter~\ref{ch:biochemical-formulas}) requires accurate elemental compositions for all metabolites. Manually entering formulas for hundreds of compounds is \textbf{error-prone and tedious}. This chapter presents \textbf{automatic formula retrieval} from the \textbf{Kyoto Encyclopedia of Genes and Genomes (KEGG)}, the largest public database of biochemical compounds and reactions.

\subsection{KEGG Database}
\label{subsec:kegg-database}

\textbf{KEGG} (\url{www.kegg.jp}) provides:

\begin{itemize}
    \item \textbf{KEGG COMPOUND}: 18,000+ metabolites with formulas, structures, pathways
    \item \textbf{KEGG REACTION}: 11,000+ biochemical reactions with stoichiometry, EC numbers
    \item \textbf{REST API}: HTTP endpoints for programmatic access (no authentication required)
\end{itemize}

\subsection{SHYpn Integration Features}
\label{subsec:shypn-kegg-integration}

\begin{enumerate}
    \item \textbf{Compound lookup}: Fetch formula by KEGG ID (e.g., ``C00031'' $\to$ \ce{C6H12O6})
    \item \textbf{Reaction parsing}: Extract stoichiometry from KEGG reaction equations
    \item \textbf{Pathway import}: Bulk import entire KEGG pathways as Bio-PN networks
    \item \textbf{Automatic enrichment}: Suggest missing cofactors (\ce{ATP}, \ce{NAD+}, \ce{H2O})
\end{enumerate}

\textbf{Benefits}:

\begin{itemize}
    \item \textbf{Speed}: Entire glycolysis pathway (10 reactions) imported in $<5$ seconds
    \item \textbf{Accuracy}: KEGG formulas are manually curated (95\%+ accuracy)
    \item \textbf{Completeness}: Automatically adds cofactors often omitted in textbook representations
\end{itemize}

\textbf{Key endpoints} (GET requests):

\textbf{Response format}: Plain text (tab-delimited fields)

\subsection{Example: Fetch Glucose}
\label{subsec:kegg-example-glucose}

\paragraph{Request:}
\begin{verbatim}
GET https://rest.kegg.jp/get/C00031
\end{verbatim}

\paragraph{Response (excerpt):}
\begin{verbatim}
ENTRY       C00031                      Compound
NAME        D-Glucose; Grape sugar; Dextrose
FORMULA     C6H12O6
EXACT_MASS  180.0634
REACTION    R00299 R00300 R00301 R00302
PATHWAY     map00010  Glycolysis / Gluconeogenesis

ENZYME      2.7.1.1   2.7.1.2

            ChEBI: 17234

\end{verbatim}

\textbf{Key fields}:

\begin{itemize}
    \item \texttt{FORMULA}: \ce{C6H12O6} (Hill notation)
    \item \texttt{REACTION}: List of reactions involving this compound
    \item \texttt{PATHWAY}: Associated pathways
    \item \texttt{DBLINKS}: Cross-references to PubChem, ChEBI
\end{itemize}

\section{KEGGConnector Class}
\label{sec:kegg-connector}

\subsection{Responsibilities}
\label{subsec:kegg-connector-responsibilities}

\begin{itemize}
    \item Send HTTP requests to KEGG REST API
    \item Parse plain-text responses
    \item Cache results (avoid redundant API calls)
    \item Handle errors (network failures, invalid IDs)
\end{itemize}

\subsection{Key Methods}
\label{subsec:kegg-connector-methods}

\begin{enumerate}
    \item \texttt{get\_compound(compound\_id)}: Fetch compound details by KEGG ID
    \begin{itemize}
        \item \textbf{Returns}: Dictionary with keys: \texttt{name}, \texttt{formula}, \texttt{exact\_mass}, \texttt{reactions}, \texttt{pathways}
        \item \textbf{Cache}: \texttt{@lru\_cache(maxsize=1000)}
    \end{itemize}
    
    \item \texttt{get\_reaction(reaction\_id)}: Fetch reaction details by KEGG ID
    \begin{itemize}
        \item \textbf{Returns}: Dictionary with keys: \texttt{name}, \texttt{definition}, \texttt{equation}, \texttt{ec\_number}, \texttt{reversible}
        \item \textbf{Cache}: \texttt{@lru\_cache(maxsize=500)}
    \end{itemize}
    
    \item \texttt{search\_compound(query)}: Search compounds by name
    \begin{itemize}
        \item \textbf{Returns}: List of matches: \texttt{[\{"id": "C00031", "name": "D-Glucose"\}, ...]}
    \end{itemize}
\end{enumerate}

\subsection{Usage Examples}
\label{subsec:kegg-connector-usage}

\paragraph{Example 1: Lookup compound by ID}

\begin{verbatim}
connector = KEGGConnector()
glucose = connector.get_compound("C00031")
# Returns:
# {
#   'name': 'D-Glucose',
#   'formula': 'C6H12O6',
#   'exact_mass': 180.0634,
#   'reactions': ['R00299', 'R00300', ...],
#   'pathways': {'map00010': 'Glycolysis / Gluconeogenesis', ...}
# }
\end{verbatim}

\paragraph{Example 2: Search compound by name}

\begin{verbatim}
results = connector.search_compound("pyruvate")
# Returns:
# [
#   {'id': 'C00022', 'name': 'Pyruvate'},
#   {'id': 'C00074', 'name': 'Phosphoenolpyruvate'},
#   ...
# ]
\end{verbatim}

\paragraph{Example 3: Fetch reaction}

\begin{verbatim}
reaction = connector.get_reaction("R00299")
# Returns:
# {
#   'name': 'hexokinase',
#   'definition': 'ATP + D-Glucose <=> ADP + D-Glucose 6-phosphate',
#   'equation': {
#     'substrates': [('C00002', 1), ('C00031', 1)],
#     'products': [('C00008', 1), ('C00092', 1)]
#   },
#   'ec_number': '2.7.1.1',
#   'reversible': True
# }
\end{verbatim}

\section{Automatic Model Enrichment}
\label{sec:model-enrichment}

\subsection{Place Formula Auto-Fill}
\label{subsec:place-formula-autofill}

\textbf{User workflow}:

\begin{enumerate}
    \item User creates a place and enters KEGG ID (e.g., ``C00031'')
    \item User clicks \textbf{``Fetch from KEGG''} button
    \item SHYpn fetches formula, name, and fills in form fields
\end{enumerate}

\textbf{Implementation}: \texttt{PlacePropertyPanel} class connects ``Fetch from KEGG'' button to \texttt{KEGGConnector.get\_compound()}.

\textbf{Benefits}:

\begin{itemize}
    \item \textbf{Speed}: No manual formula entry
    \item \textbf{Accuracy}: KEGG formulas are curated
    \item \textbf{Consistency}: Same compound always has same formula
\end{itemize}

\subsection{Reaction Import}
\label{subsec:reaction-import}

\textbf{User workflow}:

\begin{enumerate}
    \item User selects \textbf{``Import from KEGG''} menu item
    \item Dialog prompts for KEGG reaction ID (e.g., ``R00299'')
    \item SHYpn fetches reaction, creates:
    \begin{itemize}
        \item \textbf{Places} for all substrates and products (if not already present)
        \item \textbf{Transition} with appropriate name and EC number
        \item \textbf{Arcs} with stoichiometric weights
    \end{itemize}
    \item User reviews and adjusts parameters ($V_{\max}$, $K_m$)
\end{enumerate}

\paragraph{Example: Importing hexokinase (R00299)}

\textbf{Before import}: Empty model

\textbf{After import}:

\begin{itemize}
    \item \textbf{Places}:
    \begin{itemize}
        \item \texttt{p\_C00002} (ATP, \ce{C10H16N5O13P3})
        \item \texttt{p\_C00031} (D-Glucose, \ce{C6H12O6})
        \item \texttt{p\_C00008} (ADP, \ce{C10H15N5O10P2})
        \item \texttt{p\_C00092} (D-Glucose 6-phosphate, \ce{C6H13O9P})
    \end{itemize}
    
    \item \textbf{Transition}:
    \begin{itemize}
        \item \texttt{t\_R00299} (hexokinase, EC 2.7.1.1, Continuous)
    \end{itemize}
    
    \item \textbf{Arcs}:
    \begin{itemize}
        \item \texttt{p\_C00002} $\to$ \texttt{t\_R00299} (weight=1)
        \item \texttt{p\_C00031} $\to$ \texttt{t\_R00299} (weight=1)
        \item \texttt{t\_R00299} $\to$ \texttt{p\_C00008} (weight=1)
        \item \texttt{t\_R00299} $\to$ \texttt{p\_C00092} (weight=1)
    \end{itemize}
\end{itemize}

\section{Cofactor Suggestion}
\label{sec:cofactor-suggestion}

\subsection{Problem: Missing Cofactors}
\label{subsec:missing-cofactors-problem}

\textbf{KEGG reaction equations often omit common cofactors}:

\begin{itemize}
    \item Water (\ce{H2O}) added in hydrolysis
    \item Protons (\ce{H+}) released in oxidation
    \item Inorganic phosphate (\ce{Pi}) in ATP hydrolysis
\end{itemize}

\textbf{Example}: KEGG R00299 (hexokinase)

\begin{itemize}
    \item \textbf{KEGG equation}: \ce{ATP + Glucose <=> ADP + G6P}
    \item \textbf{Actual reaction}: \ce{ATP + Glucose <=> ADP + G6P + H+}
    \item \textbf{Elemental imbalance} without \ce{H+}: Left has 16 H, right has 15 H
\end{itemize}

\subsection{Cofactor Suggestion Algorithm}
\label{subsec:cofactor-algorithm}

\textbf{Algorithm}:

\begin{enumerate}
    \item Compute elemental imbalance $\Delta =$ Input elements $-$ Output elements
    \begin{itemize}
        \item Example: $\Delta = \{``H'': 1, ``O'': 0, ``C'': 0, \ldots\}$
    \end{itemize}
    
    \item Define common cofactor formulas:
    \begin{itemize}
        \item \ce{H2O}: \{``H'': 2, ``O'': 1\}
        \item \ce{H+}: \{``H'': 1\}
        \item \ce{HPO4^{2-}} (phosphate): \{``H'': 1, ``P'': 1, ``O'': 4\}
        \item \ce{NAD+}: \{``C'': 21, ``H'': 27, ``N'': 7, ``O'': 14, ``P'': 2\}
        \item \ce{NADH}: \{``C'': 21, ``H'': 29, ``N'': 7, ``O'': 14, ``P'': 2\}
        \item \ce{CO2}: \{``C'': 1, ``O'': 2\}
    \end{itemize}
    
    \item Try combinations of cofactors (linear Diophantine equation):
    \begin{itemize}
        \item Find coefficients $(n_1, n_2, \ldots)$ such that:
        \begin{equation}
        \Delta = n_1 \cdot \text{Formula}_1 + n_2 \cdot \text{Formula}_2 + \ldots
        \end{equation}
    \end{itemize}
    
    \item Rank suggestions by:
    \begin{itemize}
        \item Fewest cofactors (prefer simple)
        \item Biochemical plausibility (e.g., \ce{H+} + \ce{H2O} more likely than \ce{NAD+} alone)
    \end{itemize}
    
    \item Return top 3 suggestions
\end{enumerate}

\paragraph{Example:}

\begin{itemize}
    \item \textbf{Imbalance}: \{``H'': 1\}
    \item \textbf{Suggestion 1}: +1 \ce{H+} (rank=1.0)
    \item \textbf{Suggestion 2}: +0.5 \ce{H2O}, $-$0.5 O (rank=0.3) [less plausible]
\end{itemize}

\subsection{Example: Hexokinase Balance}
\label{subsec:hexokinase-balance-example}

\textbf{Initial reaction} (from KEGG R00299):

\begin{equation}
\ce{ATP (C10H16N5O13P3) + Glucose (C6H12O6) -> ADP (C10H15N5O10P2) + G6P (C6H13O9P)}
\end{equation}

\textbf{Elemental count}:

\begin{itemize}
    \item \textbf{Inputs}: C=16, H=28, N=5, O=19, P=4
    \item \textbf{Outputs}: C=16, H=28, N=5, O=19, P=3
    \item \textbf{Imbalance}: P=+1
\end{itemize}

\textbf{Cofactor suggester output}:

\begin{verbatim}
[
  {'cofactors': {'Pi': 1}, 'score': 1.0},
  {'cofactors': {'H+': 1, 'Pi': 1}, 'score': 0.5}
]
\end{verbatim}

\textbf{User action}: Accept suggestion $\to$ Add \ce{Pi} place and output arc

\textbf{Corrected reaction}:

\begin{equation}
\ce{ATP + Glucose -> ADP + G6P + Pi}
\end{equation}

\section{Integration with Model Validation}

\subsection{Validation Workflow}
\label{subsec:kegg-validation-workflow}

\textbf{SHYpn validation steps} (triggered on model save):

\begin{enumerate}
    \item \textbf{Structural validation} (well-formedness constraints C1--C8)
    \item \textbf{Elemental balance check} (for all transitions)
    \item \textbf{If imbalance detected}:
    \begin{itemize}
        \item Show warning dialog
        \item Display suggested cofactors
        \item User can accept, modify, or ignore
    \end{itemize}
\end{enumerate}

\textbf{User experience}:

\begin{itemize}
    \item \textbf{Green checkmark}: Model fully validated (structurally sound + elementally balanced)
    \item \textbf{Yellow warning}: Model has imbalances, but suggestions available
    \item \textbf{Red error}: Structural issues (e.g., test arc with weight $\neq$ 1)
\end{itemize}

\section{Performance and Caching}
\label{sec:kegg-performance}

\subsection{API Call Minimization}
\label{subsec:kegg-api-minimization}

\textbf{Challenge}: KEGG REST API is relatively slow (200--500 ms per request)

\textbf{Strategies}:

\begin{enumerate}
    \item \textbf{LRU cache}: \texttt{@lru\_cache} decorator on \texttt{get\_compound()} and \texttt{get\_reaction()}
    \begin{itemize}
        \item Avoids repeated fetches of same compound
        \item 1000-entry cache (sufficient for typical models)
    \end{itemize}
    
    \item \textbf{Batch requests}: When importing pathway, fetch all reactions in parallel
    
    \item \textbf{Offline mode}: Store downloaded compounds in local SQLite database
    \begin{itemize}
        \item First fetch: Download from KEGG + save to DB
        \item Subsequent: Read from DB (10$\times$ faster)
    \end{itemize}
\end{enumerate}

\textbf{Benchmark} (import glycolysis pathway):

\begin{table}[h]
\centering
\begin{tabular}{@{}lc@{}}
\toprule
\textbf{Strategy} & \textbf{Time} \\
\midrule
Without cache & 20 seconds \\
With cache & 7 seconds \\
With offline DB & 0.3 seconds \\
\bottomrule
\end{tabular}
\caption{Performance comparison for importing glycolysis pathway (10 reactions, 13 compounds)}
\label{tab:kegg-performance}
\end{table}

\subsection{Offline Database}
\label{subsec:kegg-offline-database}

\textbf{Schema} (SQLite):

\begin{verbatim}
CREATE TABLE compounds (
    kegg_id TEXT PRIMARY KEY,
    name TEXT,
    formula TEXT,
    exact_mass REAL,
    fetched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE reactions (
    kegg_id TEXT PRIMARY KEY,
    name TEXT,
    definition TEXT,
    equation TEXT,  -- JSON string
    ec_number TEXT,
    reversible BOOLEAN,
    fetched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
\end{verbatim}

\textbf{Hybrid strategy}: Check local DB first, fallback to KEGG API

\section{Limitations and Future Work}
\label{sec:kegg-limitations}

\subsection{Current Limitations}
\label{subsec:kegg-current-limitations}

\begin{enumerate}
    \item \textbf{Protonation states}: KEGG formulas are at pH 7, but protonation varies by pH
    \begin{itemize}
        \item \textbf{Impact}: Elemental counts slightly off (\ce{H+} count ambiguous)
        \item \textbf{Workaround}: Assume pH 7, ignore small H imbalances
    \end{itemize}
    
    \item \textbf{Incomplete stoichiometry}: Some KEGG reactions omit minor products
    \begin{itemize}
        \item \textbf{Example}: R00299 omits \ce{H+} released by ATP hydrolysis
        \item \textbf{Solution}: Cofactor suggester partially mitigates
    \end{itemize}
    
    \item \textbf{Rate constant data}: KEGG does not provide kinetic parameters ($V_{\max}$, $K_m$)
    \begin{itemize}
        \item \textbf{Solution}: Chapter~\ref{ch:brenda} (BRENDA integration for parameters)
    \end{itemize}
    
    \item \textbf{Pathway topology}: KEGG pathways are lists of reactions (no explicit network structure)
    \begin{itemize}
        \item \textbf{Solution}: SHYpn infers connections from shared metabolites
    \end{itemize}
\end{enumerate}

\subsection{Future Enhancements}
\label{subsec:kegg-future-enhancements}

\begin{enumerate}
    \item \textbf{ChEBI integration}: Alternative database with more detailed formulas
    \item \textbf{MetaCyc integration}: Curated metabolic pathways with better stoichiometry
    \item \textbf{Rhea integration}: Reaction database with balanced equations
    \item \textbf{Automatic rate constant estimation}: Use machine learning to predict $K_m$ from enzyme structure
\end{enumerate}
